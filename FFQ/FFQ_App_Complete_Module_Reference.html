۱. ماژول اکتیویتی اصلی (MainActivity.kt)

com/users/ffq_app/MainActivity.kt

این ماژول تنها نقطه ورود اپلیکیشن به چرخه حیات اندروید است. در معماری تک‌اکتیویتی (Single Activity)، این کلاس وظیفه راه‌اندازی متد setContent، میزبانی از پوسته تم سراسری (FFQAppTheme) و نگه داشتن وضعیت زنده تم انتخابی کاربر را بر عهده دارد.

وراثت (Inheritance): از کلاس ComponentActivity برای پشتیبانی کامل از Compose.

        setContent {

۳. ماژول هدایت‌کننده برنامه (AppNavigator.kt)

com/users/ffq_app/ui/navigation/AppNavigator.kt

این ماژول کنترل‌کننده اصلی جریان کاربری (User Flow) در اپلیکیشن است. وظایف آن عبارتند از: اعمال چیدمان راست‌به‌چپ (RTL)، مدیریت دکمه فیزیکی بازگشت، ترنزیشن متحرک بین صفحات با Crossfade و برقراری ارتباط بین صفحات مختلف.

چیدمان RTL: تزریق با LocalLayoutDirection provides LayoutDirection.Rtl
دیکشنری رشته‌ها: تزریق با LocalAppStrings provides AppStrings()
خروج ایمن: استفاده از backPressedOnce و کوروتین ۲ ثانیه‌ای در صفحه اصلی.
انیمیشن جابجایی: استفاده از Crossfade(animationSpec = tween(400))
کد پیاده‌سازی ناویگیتور:
AppNavigator.ktKotlin
@Composable
fun AppNavigator(
    themeMode: String = "light",
    onThemeModeChanged: (String) -> Unit = {}
) {
    var currentScreen: Screen by remember { mutableStateOf(Screen.Splash) }
    val activity = (LocalContext.current as? Activity)
    val context = LocalContext.current
    val scope = rememberCoroutineScope()
    var backPressedOnce by remember { mutableStateOf(false) }

    CompositionLocalProvider(
        LocalLayoutDirection provides LayoutDirection.Rtl,
        LocalAppStrings provides AppStrings()
    ) {
        BackHandler {
            when (currentScreen) {
                is Screen.Splash -> { }
                is Screen.Login -> activity?.finish()
                is Screen.AdminLogin -> currentScreen = Screen.Login
                is Screen.AdminDashboard -> currentScreen = Screen.Login
                is Screen.MainMenu -> {
                    if (backPressedOnce) {
                        activity?.finish()
                    } else {
                        backPressedOnce = true
                        Toast.makeText(context, "برای خروج، مجدداً بازگشت را بزنید", Toast.LENGTH_SHORT).show()
                        scope.launch {
                            delay(2000)
                            backPressedOnce = false
                        }
                    }
                }
                is Screen.Registration -> currentScreen = Screen.MainMenu
                is Screen.FFQQuestionnaire -> currentScreen = Screen.MainMenu
                is Screen.History -> currentScreen = Screen.MainMenu
                is Screen.PatientDetail -> currentScreen = Screen.History
                is Screen.ContactUs -> currentScreen = Screen.MainMenu
                is Screen.Settings -> currentScreen = Screen.MainMenu
            }
        }

        Crossfade(targetState = currentScreen, animationSpec = tween(400)) { screen ->
            when (screen) {
                is Screen.Splash -> SplashScreen(onFinished = { currentScreen = Screen.Login })
                is Screen.Login -> LoginScreen(
                    onLoginSuccess = { currentScreen = Screen.MainMenu },
                    onAdminClick = { currentScreen = Screen.AdminLogin }
                )
                is Screen.AdminLogin -> AdminLoginScreen(
                    onSuccess = { currentScreen = Screen.AdminDashboard },
                    onBack = { currentScreen = Screen.Login }
                )
                is Screen.AdminDashboard -> AdminDashboardScreen(
                    onBack = { currentScreen = Screen.Login },
                    onThemeModeChanged = onThemeModeChanged
                )
                is Screen.MainMenu -> MainMenuScreen(
                    onNavigate = { currentScreen = it },
                    onExitApp = { activity?.finish() },
                    onLogout = { currentScreen = Screen.Login }
                )
                is Screen.Registration -> RegistrationScreen(
                    initialData = screen.initialData,
                    onNavigateBack = { currentScreen = Screen.MainMenu },
                    onProceedToFFQ = { regData ->
                        currentScreen = Screen.FFQQuestionnaire(patientId = regData.id, registrationData = regData)
                    }
                )
                is Screen.FFQQuestionnaire -> {
                    val existing = FFQRepository.getRecord(context, screen.patientId)
                    FFQQuestionnaireScreen(
                        patientId = screen.patientId,
                        registrationData = screen.registrationData,
                        initialAnswers = existing?.answers ?: emptyMap(),
                        onNavigateBack = { currentScreen = Screen.MainMenu },
                        onSaveSuccess = { currentScreen = Screen.PatientDetail(screen.patientId) }
                    )
                }
                is Screen.History -> HistoryScreen(
                    onPatientSelected = { id -> currentScreen = Screen.PatientDetail(id) },
                    onNavigateBack = { currentScreen = Screen.MainMenu }
                )
                is Screen.PatientDetail -> PatientDetailScreen(
                    patientId = screen.patientId,
                    onNavigateBack = { currentScreen = Screen.History },
                    onEditRegistration = { reg -> currentScreen = Screen.Registration(initialData = reg) },
                    onOpenFFQ = { rec -> currentScreen = Screen.FFQQuestionnaire(rec.patientId, rec.registration) }
                )
                is Screen.ContactUs -> ContactUsScreen(onNavigateBack = { currentScreen = Screen.MainMenu })
                is Screen.Settings -> ContactUsScreen(onNavigateBack = { currentScreen = Screen.MainMenu })
            }
        }
    }
}

۴. ماژول تنظیمات پایدار (AppPrefs.kt)

این ماژول یک آبجکت کاتلین (Singleton) است که ذخیره‌سازی ماندگار تنظیمات تم و حساب‌های کاربری پرسنل را در فضای خصوصی برنامه مدیریت می‌کند.

کد کامل ماژول:
AppPrefs.ktKotlin
package com.users.ffq_app.data.local

import android.content.Context
import androidx.core.content.edit
import com.google.gson.Gson
import com.google.gson.reflect.TypeToken

object AppPrefs {
    private const val PREFS_NAME = "FFQPrefs"
    private fun getPrefs(context: Context) = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)

    fun getUsers(context: Context): List<Pair<String, String>> {
        val json = getPrefs(context).getString("allowed_users", null)
        if (json == null) {
            val initial = listOf("user" to "1234")
            saveUsers(context, initial)
            return initial
        }
        val type = object : TypeToken<List<Pair<String, String>>>() {}.type
        return try {
            Gson().fromJson(json, type) ?: emptyList()
        } catch (e: Exception) { emptyList() }
    }

    fun saveUsers(context: Context, users: List<Pair<String, String>>) {
        getPrefs(context).edit { putString("allowed_users", Gson().toJson(users)) }
    }

    fun getThemeMode(context: Context): String {
        return getPrefs(context).getString("app_theme_mode", "light") ?: "light"
    }

    fun setThemeMode(context: Context, mode: String) {
        getPrefs(context).edit { putString("app_theme_mode", mode) }
    }
}

قلب محاسباتی و ساختار داده‌ای اپلیکیشن در این فایل تعریف شده است. این ماژول دارای ۴ کلاس داده‌ای (Data Class) با پراپرتی‌های محاسباتی است.

FFQModels.ktKotlin
package com.users.ffq_app.data.model

data class FFQItem(
    val id: Int,
    val name: String,
    val portion: String,
    val notes: String = "",
    val nutrientKey: String = "",
    val nutrients: Map<String, Double> = emptyMap()
)

data class FFQAnswer(
    val day: String = "",
    val week: String = "",
    val month: String = "",
    val year: String = "",
    val userNotes: String = ""
) {
    fun isEmpty(): Boolean = day.isBlank() && week.isBlank() && month.isBlank() && year.isBlank() && userNotes.isBlank()
    
    // محاسبه ضریب دریافت روزانه بر اساس اولین فیلد پر شده
    fun toDailyMultiplier(): Double {
        val d = day.toDoubleOrNull()
        if (d != null && d > 0) return d
        val w = week.toDoubleOrNull()
        if (w != null && w > 0) return w / 7.0
        val m = month.toDoubleOrNull()
        if (m != null && m > 0) return m / 30.42
        val y = year.toDoubleOrNull()
        if (y != null && y > 0) return y / 365.0
        return 0.0
    }
}

data class RegistrationData(
    val id: String = "",
    val diagnosisMO: String = "",
    val marital: Int? = null,
    val otherNarcotics: String = "",
    val ghelian: String = "",
    val cigar: String = "",
    val aspirin: Int? = null,
    val levotyroxin: Int? = null,
    val statin: String = "",
    val vitD: Int? = null,
    val multivitamin: Int? = null,
    val diabetes: Int? = null,
    val cvd: Int? = null,
    val bp: Int? = null,
    val migrane: Int? = null,
    val gid: Int? = null,
    val cancer: Int? = null,
    val weightKg: String = "",
    val heightCm: String = "",
    val gender: Int? = null,
    val age: String = "",
    val education: Int? = null,
    val job: Int? = null,
    val familySize: String = "",
    val kids: String = "",
    val houseOwnership: Int? = null,
    val ipaq: Map<Int, String> = emptyMap(),
    val timestamp: Long = System.currentTimeMillis()
) {
    val bmi: Double? get() {
        val w = weightKg.toDoubleOrNull() ?: return null
        val h = heightCm.toDoubleOrNull() ?: return null
        if (h <= 0) return null
        val hm = h / 100.0
        return w / (hm * hm)
    }

    val ipaqMet: Double get() {
        fun pair(p1: Int, p2: Int, factor: Double): Double {
            val v1 = ipaq[p1]?.toDoubleOrNull() ?: 0.0
            val v2 = ipaq[p2]?.toDoubleOrNull() ?: 0.0
            return v1 * v2 * factor
        }
        return pair(2, 3, 8.0) + pair(4, 5, 4.0) + pair(6, 7, 3.3) +
               pair(8, 9, 0.0) + pair(10, 11, 4.0) + pair(12, 13, 3.3) +
               pair(14, 15, 8.0) + pair(16, 17, 4.0) + pair(18, 19, 3.3) +
               pair(20, 21, 3.3) + pair(22, 23, 8.0) + pair(24, 25, 4.0)
    }
}

۷. ماژول مخزن پرونده‌ها و دیتابیس (FFQRepository.kt)

com/users/ffq_app/util/FFQRepository.kt

این ماژول لایه انتزاعی پایگاه داده است. کش درون‌حافظه‌ای اقلام غذایی (In-Memory Caching) را انجام می‌دهد و عملیات ذخیره، خواندن، لیست‌گیری و حذف پرونده‌ها را روی دیسک مدیریت می‌کند.

FFQRepository.ktKotlin
object FFQRepository {
    private var cachedItems: List<FFQItem>? = null

    fun loadFFQItems(context: Context): List<FFQItem> {
        cachedItems?.let { return it }
        return try {
            val jsonString = context.assets.open("ffq_database.json").bufferedReader().use { it.readText() }
            val type = object : TypeToken<List<Map<String, Any>>>() {}.type
            val rawList: List<Map<String, Any>> = Gson().fromJson(jsonString, type)
            val list = rawList.map { m ->
                val id = (m["id"] as? Number)?.toInt() ?: 0
                val name = m["name"] as? String ?: ""
                val portion = m["portion"] as? String ?: ""
                val notes = m["notes"] as? String ?: ""
                val nutrientKey = m["nutrient_key"] as? String ?: ""
                @Suppress("UNCHECKED_CAST")
                val rawNutrients = m["nutrients"] as? Map<String, Any> ?: emptyMap()
                val nutrients = rawNutrients.mapValues { (_, v) -> (v as? Number)?.toDouble() ?: 0.0 }
                FFQItem(id, name, portion, notes, nutrientKey, nutrients)
            }.sortedBy { it.id }
            cachedItems = list
            list
        } catch (e: Exception) { emptyList() }
    }

    private fun getStorageDir(context: Context): File {
        val dir = File(context.getExternalFilesDir(null), "FFQRecords")
        if (!dir.exists()) dir.mkdirs()
        return dir
    }

    fun saveRecord(context: Context, record: PatientRecord): Boolean {
        return try {
            val dir = getStorageDir(context)
            val file = File(dir, "${record.patientId}_record.json")
            file.writeText(Gson().toJson(record))
            true
        } catch (e: Exception) { false }
    }

    fun getAllRecords(context: Context): List<PatientRecord> {
        val dir = getStorageDir(context)
        val files = dir.listFiles { f -> f.extension == "json" } ?: return emptyList()
        return files.mapNotNull { f ->
            try { Gson().fromJson(f.readText(), PatientRecord::class.java) } catch (e: Exception) { null }
        }.sortedByDescending { it.timestamp }
    }
}

۹. ماژول صفحه اسپلش با شبیه‌ساز بوت ویندوز XP (SplashScreen.kt)

com/users/ffq_app/ui/screens/SplashScreen.kt

این ماژول وظیفه نمایش صفحه ورودی و هویت بصری نرم‌افزار را برعهده دارد. در این صفحه، یک انیمیشن ملایم نبض (Pulse) برای لوگوی گرد اپلیکیشن و یک لودر اختصاصی بازآفرینی‌شده به سبک کلاسیک بوت ویندوز XP (Windows XP Boot Progress Bar) با سه خانه متحرک و رنگ‌های گرادیان سبز پاستیلی پیاده‌سازی شده است.

زمان‌سنج انتقال خودکار: اجرای کوروتین LaunchedEffect(Unit) با تاخیر ۲۸۰۰ میلی‌ثانیه‌ای (۲.۸ ثانیه) و فراخوانی کالبک onFinished() جهت ورود به صفحه لاگین.

private fun formatFreq(value: Double): String {
    if (value <= 0.0) return ""
    val rounded = Math.round(value * 100.0) / 100.0
    return if (rounded % 1.0 == 0.0) {
        rounded.toLong().toString()
    } else {
        String.format(java.util.Locale.US, "%.2f", rounded).trimEnd('0').trimEnd('.')
    }
}

// تبدیل هوشمند فرکانس مصرف با پر شدن فیلد سال:
FrequencyInput(label = "سال", value = answer.year) { input ->
    if (input.isBlank()) {
        onAnswerChange(FFQAnswer(userNotes = answer.userNotes))
    } else {
        val y = input.toDoubleOrNull()
        if (y != null && y >= 0) {
            val d = formatFreq(y / 365.0)
            val w = formatFreq(y / 52.14)
            val m = formatFreq(y / 12.0)
            onAnswerChange(FFQAnswer(day = d, week = w, month = m, year = input, userNotes = answer.userNotes))
        } else {
            onAnswerChange(answer.copy(year = input))
        }
    }

۱۸. ماژول موتور مستقل تولید اکسل دوصفحه‌ای (ExcelExporter.kt)

com/users/ffq_app/util/ExcelExporter.kt

یکی از برجسته‌ترین و پیشرفته‌ترین بخش‌های مهندسی این پروژه، موتور تولید اکسل کاملاً بومی و مستقل (Zero-Dependency OpenXML Engine) است. این ماژول بدون اتکا به کتابخانه‌های سنگین و ناسازگار خارجی مانند Apache POI (که حجم APK را بیش از ۱۵ مگابایت افزایش می‌دهند و منجر به کمبود حافظه OOM می‌شوند)، مستقیماً با جریان فشرده‌سازی بایت‌های زیپ (ZipOutputStream) ساختار استاندارد ECMA-376 یک فایل معتبر .xlsx با دو برگه تفکیک‌شده، استایل‌های رنگی اختصاصی و جهت‌گیری راست‌به‌چپ (RTL) را خلق می‌کند.

تولید فایل دو برگه‌ای (2-Sheet): برگه اول: ۵۲ ستون دموگرافیک، بیماری، دارو، تن‌سنجی و IPAQ. برگه دوم: ماتریس بسامد مصرف اقلام خوراکی با ملاحظات بیمار.
جهت‌گیری راست‌به‌چپ (RTL): تزریق تگ <sheetView rightToLeft="1" workbookViewId="0"/> در ساختار XML هر برگه جهت چیدمان اصولی فارسی.
مبدل پایه‌۲۶ نام ستون‌ها: الگوریتم getColName(colIdx) برای تبدیل شاخص‌های عددی به برچسب‌های حروفی اکسل (مانند ۱ به A، ۲۶ به Z، ۲۷ به AA).
اشتراک‌گذاری امن سیستمی: استفاده از FileProvider.getUriForFile و اینتنت‌های ACTION_SEND و ACTION_VIEW برای باز کردن یا اشتراک در پیام‌رسان‌ها.
کد معماری موتور زیپ و ساختار OpenXML:
ExcelExporter.ktKotlin
package com.users.ffq_app.util

import android.content.Context
import android.content.Intent
import android.os.Environment
import androidx.core.content.FileProvider
import com.users.ffq_app.data.model.PatientRecord
import java.io.File
import java.io.FileOutputStream
import java.util.zip.ZipEntry
import java.util.zip.ZipOutputStream

data class XlsxCell(val value: Any?, val isHeader: Boolean = false)
data class XlsxSheet(val title: String, val rows: List<List<XlsxCell>>)

object ExcelExporter {

    private fun escapeXml(str: String): String {
        return str.replace("&", "&")
            .replace("<", "<")
            .replace(">", ">")
            .replace(""", """)
            .replace("'", "'")
    }

    private fun getColName(colIdx: Int): String {
        var col = colIdx
        val sb = StringBuilder()
        while (col > 0) {
            val rem = (col - 1) % 26
            sb.append(('A'.code + rem).toChar())
            col = (col - 1) / 26
        }
        return sb.reverse().toString()
    }

    fun buildXlsxFile(sheets: List<XlsxSheet>, outputFile: File) {
        outputFile.parentFile?.mkdirs()
        ZipOutputStream(FileOutputStream(outputFile)).use { zos ->
            // ۱. [Content_Types].xml
            zos.putNextEntry(ZipEntry("[Content_Types].xml"))
            zos.write(buildContentTypesXml(sheets.size).toByteArray(Charsets.UTF_8))
            zos.closeEntry()

            // ۲. _rels/.rels
            zos.putNextEntry(ZipEntry("_rels/.rels"))
            zos.write(buildRelsXml().toByteArray(Charsets.UTF_8))
            zos.closeEntry()

            // ۳. xl/workbook.xml و xl/_rels/workbook.xml.rels
            zos.putNextEntry(ZipEntry("xl/workbook.xml"))
            zos.write(buildWorkbookXml(sheets).toByteArray(Charsets.UTF_8))
            zos.closeEntry()

            // ۴. استایل‌های جدول با رنگ هدر سبز پاستیلی و کادر خاکستری
            zos.putNextEntry(ZipEntry("xl/styles.xml"))
            zos.write(buildStylesXml().toByteArray(Charsets.UTF_8))
            zos.closeEntry()

            // ۵. ایجاد برگه‌ها با جهت‌گیری RTL
            for ((idx, sheet) in sheets.withIndex()) {
                zos.putNextEntry(ZipEntry("xl/worksheets/sheet${idx + 1}.xml"))
                zos.write(buildWorksheetXml(sheet).toByteArray(Charsets.UTF_8))
                zos.closeEntry()
            }
        }
    }
}

package com.users.ffq_app.ui.theme

import android.app.Activity
import androidx.compose.foundation.isSystemInDarkTheme
import androidx.compose.material3.*
import androidx.compose.runtime.Composable
import androidx.compose.runtime.SideEffect
import androidx.compose.ui.graphics.toArgb
import androidx.compose.ui.platform.LocalView
import androidx.core.view.WindowCompat

@Composable
fun FFQAppTheme(
    themeMode: String = "light",
    content: @Composable () -> Unit
) {
    val isSystemDark = isSystemInDarkTheme()
    val isDark = when (themeMode) {
        "dark" -> true
        "light" -> false
        else -> isSystemDark
    }
    val colorScheme = if (isDark) DarkColorScheme else LightColorScheme
    val view = LocalView.current
    if (!view.isInEditMode) {
        SideEffect {
            val window = (view.context as Activity).window
            window.statusBarColor = colorScheme.primary.toArgb()
            WindowCompat.getInsetsController(window, view).isAppearanceLightStatusBars = false
        }
    }

    MaterialTheme(
        colorScheme = colorScheme,
        typography = Typography,
        content = content
    )
}
